/******************************************************* * APLIKASI MEDICATION ERROR - DEPO FARMASI * Backend: Google Apps Script *******************************************************/ // ====== KONFIGURASI ====== const SPREADSHEET_ID = 'ISI_DENGAN_ID_SPREADSHEET_ANDA'; // Ganti dengan ID Spreadsheet const SHEET_NAME = 'MedicationError'; // Email koordinator tiap depo (ganti sesuai kebutuhan) const COORDINATORS = { 'Depo Farmasi Rawat Jalan': 'koordinator.rajal@rs-anda.com', 'Depo Farmasi Rawat Inap' : 'koordinator.ranap@rs-anda.com', 'Depo Farmasi IGD' : 'koordinator.igd@rs-anda.com' }; // ====== WEB APP ====== function doGet() { return HtmlService.createHtmlOutputFromFile('Index') .setTitle('Pelaporan Medication Error') .addMetaTag('viewport', 'width=device-width, initial-scale=1') .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL); } // ====== UTIL ====== function getSheet_() { const ss = SpreadsheetApp.openById(SPREADSHEET_ID); let sheet = ss.getSheetByName(SHEET_NAME); if (!sheet) { sheet = ss.insertSheet(SHEET_NAME); sheet.appendRow([ 'Timestamp', 'ID Laporan', 'Depo', 'Tanggal Kejadian', 'Nama Pasien', 'Nama Petugas', 'Jenis Kesalahan', 'Status', 'Tindak Lanjut' ]); sheet.getRange(1, 1, 1, 9).setFontWeight('bold').setBackground('#1a73e8').setFontColor('#fff'); sheet.setFrozenRows(1); } return sheet; } // ====== SIMPAN DATA ====== function saveMedicationError(data) { try { if (!data.depo || !data.tanggal || !data.namaPasien || !data.namaPetugas || !data.jenisKesalahan) { return { success: false, message: 'Semua field wajib diisi!' }; } const sheet = getSheet_(); const id = 'ME-' + Utilities.formatDate(new Date(), 'Asia/Jakarta', 'yyyyMMddHHmmss'); sheet.appendRow([ new Date(), id, data.depo, data.tanggal, data.namaPasien, data.namaPetugas, data.jenisKesalahan, 'Baru', '' // Tindak lanjut ]); // Kirim notifikasi email ke koordinator notifyCoordinator_(data, id); return { success: true, message: 'Laporan berhasil disimpan dengan ID: ' + id, id: id }; } catch (e) { return { success: false, message: 'Error: ' + e.toString() }; } } // ====== NOTIFIKASI KOORDINATOR ====== function notifyCoordinator_(data, id) { try { const email = COORDINATORS[data.depo]; if (!email) return; const subject = '⚠️ [MEDICATION ERROR] ' + data.depo + ' - ' + data.jenisKesalahan; const body = 'LAPORAN MEDICATION ERROR BARU\n' + '=====================================\n\n' + 'ID Laporan : ' + id + '\n' + 'Depo : ' + data.depo + '\n' + 'Tanggal Kejadian: ' + data.tanggal + '\n' + 'Nama Pasien : ' + data.namaPasien + '\n' + 'Nama Petugas : ' + data.namaPetugas + '\n' + 'Jenis Kesalahan : ' + data.jenisKesalahan + '\n\n' + '=====================================\n' + 'Mohon SEGERA ditindaklanjuti.\n\n' + 'Aplikasi Pelaporan Medication Error'; MailApp.sendEmail(email, subject, body); } catch (e) { console.error('Gagal kirim email: ' + e.toString()); } } // ====== AMBIL RIWAYAT ====== function getHistory() { try { const sheet = getSheet_(); const lastRow = sheet.getLastRow(); if (lastRow <= 1) return []; const startRow = Math.max(2, lastRow - 49); // Ambil 50 terakhir const rows = sheet.getRange(startRow, 1, lastRow - startRow + 1, 9).getValues(); rows.reverse(); return rows.map(function(r) { return { timestamp: r[0] ? Utilities.formatDate(new Date(r[0]), 'Asia/Jakarta', 'dd/MM/yyyy HH:mm') : '', id: r[1], depo: r[2], tanggal: r[3] instanceof Date ? Utilities.formatDate(r[3], 'Asia/Jakarta', 'dd/MM/yyyy') : r[3], namaPasien: r[4], namaPetugas: r[5], jenisKesalahan: r[6], status: r[7], tindakLanjut: r[8] }; }); } catch (e) { return []; } } // ====== UPDATE STATUS / TINDAK LANJUT ====== function updateStatus(id, status, tindakLanjut) { try { const sheet = getSheet_(); const data = sheet.getDataRange().getValues(); for (let i = 1; i < data.length; i++) { if (data[i][1] === id) { sheet.getRange(i + 1, 8).setValue(status); sheet.getRange(i + 1, 9).setValue(tindakLanjut || ''); return { success: true, message: 'Status berhasil diupdate' }; } } return { success: false, message: 'ID tidak ditemukan' }; } catch (e) { return { success: false, message: e.toString() }; } }
