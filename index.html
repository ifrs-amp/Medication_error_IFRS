/** * LAPORAN MEDICATION ERROR — Google Apps Script backend (mode API JSON) * ----------------------------------------------------------------------- * Arsitektur ini memisahkan frontend dan backend: * - HTML (form + riwayat) di-hosting di GitHub Pages (file terpisah). * - Apps Script ini HANYA berfungsi sebagai API JSON yang dipanggil lewat * fetch() dari HTML tersebut. Ini menghindari error "No HTML file * named index was found" karena Apps Script tidak lagi merender HTML. * * CARA SETUP: * 1. Buat Google Spreadsheet baru (kosong). * 2. Extensions > Apps Script, hapus isi bawaan, tempel seluruh kode ini. * 3. Jalankan fungsi "setupSheet" sekali (pilih di dropdown lalu Run, * izinkan akses saat diminta) untuk membuat sheet "Laporan" + header. * 4. Deploy > New deployment > tipe "Web app": * - Execute as: Me * - Who has access: Anyone * (harus "Anyone", bukan "Anyone with Google account", supaya bisa * dipanggil dari halaman GitHub Pages tanpa login Google). * 5. Salin URL hasil deploy (diakhiri /exec) — tempel ke konstanta * SCRIPT_URL di file index.html (GitHub Pages). * 6. Setiap kali kode di sini diubah, deploy ulang sebagai "New version" * (Deploy > Manage deployments > ikon pensil > Version: New version). * * CATATAN CORS: frontend mengirim POST dengan Content-Type "text/plain" * secara sengaja (bukan "application/json"). Ini membuat request menjadi * "simple request" sehingga browser TIDAK mengirim preflight OPTIONS * (yang tidak didukung Apps Script). Jangan ubah Content-Type di HTML * menjadi application/json, nanti fetch akan gagal karena CORS. */ var SHEET_NAME = 'Laporan'; var HEADERS = ['Timestamp', 'Depo', 'Jenis Kesalahan', 'Nama Pasien', 'Nama Petugas', 'Tanggal Kejadian', 'Kronologi', 'Jam Kejadian']; /** GET /exec?action=list -> daftar seluruh laporan */ function doGet(e) { try { var action = (e && e.parameter && e.parameter.action) || 'list'; if (action === 'list') { return jsonResponse_({ ok: true, data: getReports() }); } return jsonResponse_({ ok: false, error: 'Aksi tidak dikenal: ' + action }); } catch (err) { return jsonResponse_({ ok: false, error: 'Terjadi kesalahan server: ' + err.message }); } } /** * POST /exec dengan body JSON. * - { ...field laporan } -> menyimpan laporan baru (default) * - { action: 'delete', timestamp } -> menghapus satu laporan berdasarkan timestamp */ function doPost(e) { try { if (!e || !e.postData || !e.postData.contents) { throw new Error('Tidak ada data yang dikirim.'); } var data = JSON.parse(e.postData.contents); var action = data && data.action ? data.action : 'submit'; if (action === 'delete') { deleteReport(data.timestamp); } else { submitReport(data); } return jsonResponse_({ ok: true }); } catch (err) { return jsonResponse_({ ok: false, error: err.message }); } } function jsonResponse_(obj) { return ContentService .createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); } /** Mengambil (atau membuat) sheet "Laporan" lengkap dengan header. */ function getOrCreateSheet_() { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheetByName(SHEET_NAME); if (!sheet) { sheet = ss.insertSheet(SHEET_NAME); sheet.appendRow(HEADERS); sheet.setFrozenRows(1); } else if (sheet.getLastRow() === 0) { // Sheet sudah ada tapi kosong (belum ada header) — pastikan header tetap dibuat. sheet.appendRow(HEADERS); sheet.setFrozenRows(1); } return sheet; } /** * SETUP MANUAL — jalankan fungsi ini SATU KALI dari editor Apps Script * (pilih "setupSheet" di dropdown fungsi lalu klik Run) untuk menyiapkan * nama sheet dan header sebelum web app mulai dipakai. Aman dijalankan * berkali-kali; tidak menghapus data yang sudah ada. */ function setupSheet() { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheetByName(SHEET_NAME); if (!sheet) { var sheets = ss.getSheets(); var emptySheet = sheets.length === 1 && sheets[0].getLastRow() === 0 ? sheets[0] : null; if (emptySheet) { emptySheet.setName(SHEET_NAME); sheet = emptySheet; } else { sheet = ss.insertSheet(SHEET_NAME); } } var headerRange = sheet.getRange(1, 1, 1, HEADERS.length); headerRange.setValues([HEADERS]); headerRange.setFontWeight('bold'); headerRange.setBackground('#e6f4f1'); sheet.setFrozenRows(1); sheet.setColumnWidth(1, 150); sheet.setColumnWidth(2, 190); sheet.setColumnWidth(3, 190); sheet.setColumnWidth(4, 160); sheet.setColumnWidth(5, 160); sheet.setColumnWidth(6, 130); sheet.setColumnWidth(7, 350); sheet.setColumnWidth(8, 90); ss.toast('Sheet "' + SHEET_NAME + '" dan header sudah siap.', 'Setup selesai', 5); } /** * data: {depo, jenis, pasien, petugas, tanggal, kronologi} * Semua field wajib diisi dan tidak boleh hanya berisi spasi. */ function submitReport(data) { var required = ['depo', 'jenis', 'pasien', 'petugas', 'tanggal', 'jam', 'kronologi']; for (var i = 0; i < required.length; i++) { var key = required[i]; if (!data || data[key] === undefined || data[key] === null || String(data[key]).trim() === '') { throw new Error('Semua kolom wajib diisi.'); } } var sheet = getOrCreateSheet_(); sheet.appendRow([ new Date(), String(data.depo).trim(), String(data.jenis).trim(), String(data.pasien).trim(), String(data.petugas).trim(), String(data.tanggal).trim(), String(data.kronologi).trim(), String(data.jam).trim() ]); return { ok: true }; } /** * Menghapus satu laporan berdasarkan timestamp (ISO string) miliknya. * Timestamp dipakai sebagai penanda unik karena setiap laporan baru * selalu mendapat waktu simpan (kolom A) yang berbeda hingga milidetik. */ function deleteReport(timestamp) { if (!timestamp) { throw new Error('Timestamp laporan diperlukan untuk menghapus.'); } var sheet = getOrCreateSheet_(); var lastRow = sheet.getLastRow(); if (lastRow < 2) { throw new Error('Tidak ada laporan untuk dihapus.'); } var timestamps = sheet.getRange(2, 1, lastRow - 1, 1).getValues(); for (var i = 0; i < timestamps.length; i++) { var cell = timestamps[i][0]; var cellIso = cell instanceof Date ? cell.toISOString() : String(cell); if (cellIso === timestamp) { sheet.deleteRow(i + 2); // +2 karena baris 1 = header, array mulai dari 0 return; } } throw new Error('Laporan tidak ditemukan (mungkin sudah dihapus sebelumnya).'); } /** Mengambil seluruh riwayat laporan, diurutkan dari tanggal kejadian terbaru. */ function getReports() { var sheet = getOrCreateSheet_(); var lastRow = sheet.getLastRow(); if (lastRow < 2) return []; var values = sheet.getRange(2, 1, lastRow - 1, HEADERS.length).getValues(); var reports = values .filter(function (row) { // Lewati baris yang benar-benar kosong (mis. sisa baris terhapus manual). return row.some(function (cell) { return cell !== '' && cell !== null; }); }) .map(function (row) { return { timestamp: row[0] instanceof Date ? row[0].toISOString() : String(row[0]), depo: row[1], jenis: row[2], pasien: row[3], petugas: row[4], tanggal: row[5] instanceof Date ? Utilities.formatDate(row[5], Session.getScriptTimeZone(), 'yyyy-MM-dd') : String(row[5]), kronologi: row[6], jam: row[7] instanceof Date ? Utilities.formatDate(row[7], Session.getScriptTimeZone(), 'HH:mm') : String(row[7] || '') }; }); reports.sort(function (a, b) { return (b.tanggal || '').localeCompare(a.tanggal || '') || (b.timestamp || '').localeCompare(a.timestamp || ''); }); return reports; }
